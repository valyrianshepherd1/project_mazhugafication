\section{Lecture 8}
\subsection*{Analytic Functions}

\begin{definition}\label{def:8.1}
Let $n\in\N\cup\{0\}$ and let $I\subseteq\R$ be an open interval. Then, a function $f:I\to\R$ is said to be
\begin{enumerate}[label=\alph*)]
    \item\label{item:definition:8.1:a} $n$-\emph{differentiable} on a set $E\subseteq I$ if the derivatives
    \[
    f^0=f,\ f',\ f'',\ldots,\ f^{(n)}
    \]
    exist at every point of $E$;
    \item\label{item:definition:8.1:b} of \emph{differentiable class} $C^n(E)$ if the derivatives
    \[
    f^0=f,\ f',\ f'',\ldots,\ f^{(n)}
    \]
    exist and are continuous at every point of $E$;
    \item\label{item:definition:8.1:c} \emph{smooth} (or \emph{infinitely differentiable}) on $E$, denoted $f\in C^\infty(E)$, if it has derivatives of all orders at every point of $E$.
\end{enumerate}
It ought to be clear that (see Calculus-1)
\[
C^n(E)\subset C^{n-1}(E),\qquad C^\infty(E)=\bigcap_{n=0}^{\infty}C^n(E).
\]
\end{definition}

\begin{definition}\label{def:8.2}
Let $I\subseteq\R$ be an open interval and let a function $f:I\to\R$ be smooth at a point $\lambda\in I$ (i.e. $f\in C^\infty(\{\lambda\})$). Then, the \emph{Taylor series} of the function $f(x)$ at the point $\lambda$ (denoted $T_{f,\lambda}(x)$) is the power series of the form
\[
T_{f,\lambda}(x)
=\sum_{k=0}^{\infty}\frac{f^{(k)}(\lambda)}{k!}(x-\lambda)^k
=f(\lambda)+f'(\lambda)(x-\lambda)+\frac{f''(\lambda)}{2!}(x-\lambda)^2+\cdots.
\tag{1}\label{eq:8:1:1}
\]
Coefficients $\dfrac{f^{(k)}(\lambda)}{k!}$ are called the \emph{Taylor coefficients}.
\end{definition}

\begin{definition}\label{def:8.3}
Let $I\subseteq\R$ be an open interval. Then a function $f:I\to\R$ is said to be \emph{analytic at a point} $\lambda\in I$, denoted $f\in C^\omega(\{\lambda\})$, if there is a sequence $\{a_k\}_{k=0}^{\infty}\subset\R$ and $\delta>0$ such that
\[
f(x)=\sum_{k=0}^{\infty}a_k(x-\lambda)^k,
\qquad \forall x\in(\lambda-\delta,\lambda+\delta).
\]
That is, a function $f$ is analytic at a point $\lambda\in I$ if it can be given by a convergent power series centered at $\lambda$ in some neighbourhood of $\lambda$.

A function $f$ is \emph{analytic on an open interval} $J\subseteq I$ if it is analytic at every point $\lambda\in J$.
\end{definition}

\begin{claim}[Any Analytic Function is Smooth]\label{claim:8.1}
Let $I\subseteq\R$ be an open interval.
If a function $f:I\to\R$ is analytic at a point $\lambda\in I$ (i.e. $f\in C^\omega(\{\lambda\})$), then:
\begin{enumerate}
    \item\label{item:claim:8.1:1} $f$ is smooth at $\lambda$ (i.e. $f\in C^\infty(\{\lambda\})$);
    \item\label{item:claim:8.1:2}
    \[
    f(x)=T_{f,\lambda}(x),\qquad \forall x\in(\lambda-\delta,\lambda+\delta),
    \]
    for some $\delta>0$, where $T_{f,\lambda}(x)$ is the Taylor series of the function $f(x)$ at the point $\lambda$ (see \hyperref[def:8.2]{Def 8.2}).
\end{enumerate}
\end{claim}

\begin{proof}
1) Due to \hyperref[def:8.3]{Def 8.3}, we have
\[
f(x)=\sum_{n=0}^{\infty}a_n(x-\lambda)^n,
\qquad \forall x\in(\lambda-\delta,\lambda+\delta),
\tag{1}\label{eq:8:1:2}
\]
for some $\delta>0$.
Therefore, due to \hyperref[def:7.2]{Def 7.2}, the series $\sum_{n=0}^{\infty}a_n(x-\lambda)^n$ has a positive radius of convergence $R\geq\delta>0$.
Thus, due to \hyperref[claim:7.6]{Claim 7.6}, \hyperref[eq:8:1:2]{Equality (1)} implies that, for every $k\in\N\cup\{0\}$ and every $x\in(\lambda-\delta,\lambda+\delta)$, we have
\[
f^{(k)}(x)
=\sum_{n=k}^{\infty}n(n-1)\cdots(n-k+1)a_n(x-\lambda)^{n-k}.
\]
If we substitute $x=\lambda$, then
\[
f^{(k)}(\lambda)=k!a_k,
\qquad a_k=\frac{f^{(k)}(\lambda)}{k!},
\qquad \forall k\in\N\cup\{0\}.
\tag{2}\label{eq:8:2:1}
\]
Thus, due to \hyperref[def:8.1]{Def 8.1}.c), $f$ is smooth on $E=\{\lambda\}$.

2) Note that, $\forall x\in(\lambda-\delta,\lambda+\delta)$, we have
\[
f(x)
\overset{1}{=}\sum_{k=0}^{\infty}a_k(x-\lambda)^k
\overset{2}{=}\sum_{k=0}^{\infty}\frac{f^{(k)}(\lambda)}{k!}(x-\lambda)^k
\overset{3}{=}T_{f,\lambda}(x).
\]
Here 1 holds due to \hyperref[eq:8:1:2]{Equality (1)}, 2 holds due to \hyperref[eq:8:2:1]{Equality (2)}, and 3 holds due to \hyperref[def:8.2]{Def 8.2} and since $f\in C^\infty(\{\lambda\})$ due to \hyperref[item:claim:8.1:1]{Item 1}.
\end{proof}

\begin{claim}[Example of non-analytic smooth function]\label{claim:8.2}
Let $\widetilde\tau:\R\to\R$ be defined as
\[
\widetilde\tau(x)=
\begin{cases}
e^{-1/x},&x>0,\\
0,&x\leq0.
\end{cases}
\]
Then:
\begin{enumerate}
    \item\label{item:claim:8.2:1}
    \[
    \widetilde\tau^{(n)}(x)=
    \begin{cases}
    \dfrac{P_n(x)}{x^{2n}}\,\widetilde\tau(x),&x>0,\\
    0,&x\leq0,
    \end{cases}
    \qquad \forall n\in\N,
    \tag{3}\label{eq:8:3:1}
    \]
    where $P_n(x)$ is a polynomial of degree $n-1$ given recursively by $P_1(x)=1$ and
    \[
    P_{n+1}(x)=x^2P_n'(x)-(2nx-1)P_n(x),\qquad \forall n\geq1;
    \]
    \item\label{item:claim:8.2:2} $\widetilde\tau\in C^\infty(\R)$ and $\widetilde\tau^{(n)}(0)=0$, $\forall n\in\N\cup\{0\}$;
    \item\label{item:claim:8.2:3} $\widetilde\tau$ is not analytic at $\lambda=0$.
\end{enumerate}
\end{claim}

\begin{proof}
1) and 2) One can prove by math. induction on $n\in\N$.
We leave it as an exercise for the reader.

3) Let us assume that $\widetilde\tau(x)$ is analytic at $\lambda=0$.
Then, for some $\delta>0$ and $\forall x\in(0-\delta,0+\delta)=(-\delta,\delta)$ we have
\[
\widetilde\tau(x)
=T_{\widetilde\tau,0}(x)
=\sum_{k=0}^{\infty}\frac{\widetilde\tau^{(k)}(0)}{k!}(x-0)^k
=\sum_{k=0}^{\infty}0\cdot x^k
=0.
\tag{4}\label{eq:8:4:1}
\]
The first equality holds due to \hyperref[item:claim:8.1:2]{Item 2)} of \hyperref[claim:8.1]{Claim 8.1} (since we assumed that $\widetilde\tau\in C^\omega(\{0\})$), the second due to \hyperref[def:8.2]{Def 8.2} and since $\widetilde\tau\in C^\infty(\{0\})\subset C^\infty(\R)$, and the third since $\widetilde\tau^{(k)}(0)=0$, $\forall k\in\N\cup\{0\}$.

\hyperref[eq:8:4:1]{Equality (4)} contradicts the definition of $\widetilde\tau$ (indeed, $\forall x>0$ we have $\widetilde\tau(x)=e^{-1/x}>0$). Thus, our assumption is false and $\widetilde\tau(x)$ is not analytic at $\lambda=0$.
\end{proof}

Later we will need the following well-known fact.

\begin{claim}[The Taylor Formula]\label{claim:8.3}
Let $I\subseteq\R$ be an open interval, let $f:I\to\R$ be an $(n+1)$-differentiable on $I$ function (see \hyperref[def:8.1]{Def 8.1}.a)), and let $\lambda\in I$.
Then, for each $x\in I$, there is a point $\beta$ located between $\lambda$ and $x$ such that
\[
f(x)=\sum_{k=0}^{n}\frac{f^{(k)}(\lambda)}{k!}(x-\lambda)^k+R_n(x),
\tag{5}\label{eq:8:5:1}
\]
where the remainder term $R_n(x)$ can be written as
\begin{itemize}
    \item
    \[
    R_n(x)=\frac{f^{(n+1)}(\beta)}{(n+1)!}(x-\lambda)^{n+1}
    \qquad [\text{the Lagrange form of }R_n(x)]
    \]
    or
    \item
    \[
    R_n(x)=\frac{f^{(n+1)}(\beta)}{n!}(x-\beta)^n(x-\lambda)
    \qquad [\text{the Cauchy form of }R_n(x)].
    \]
\end{itemize}
\end{claim}

\begin{proof}
See Calculus-1.
\end{proof}

\begin{claim}[Criterion for Analyticity]\label{claim:8.4}
Let $I\subseteq\R$ be an open interval and let $f:I\to\R$ be a smooth function on $I$ (i.e. $f\in C^\infty(I)$).
Then, the following two conditions are equivalent:
\begin{enumerate}
    \item\label{item:claim:8.4:1} $f$ is analytic at $\lambda\in I$ (i.e. $f\in C^\omega(\{\lambda\})$);
    \item\label{item:claim:8.4:2} there exist constants $L>d>0$ and $C>0$ such that
    \[
    (\lambda-d,\lambda+d)\subset I
    \quad\text{and}\quad
    |f^{(k)}(x)|\leq C\cdot\frac{k!}{L^k}
    \]
    for all $k\in\N\cup\{0\}$ and all $x\in(\lambda-d,\lambda+d)$.
\end{enumerate}
\end{claim}

\begin{proof}
$2)\Rightarrow1)$ Due to \hyperref[def:8.3]{Def 8.3}, it suffices to prove that
\[
\exists\delta>0\quad \forall x\in(\lambda-\delta,\lambda+\delta)
\quad\to\quad f(x)=T_{f,\lambda}(x),
\tag{6}\label{eq:8:6:1}
\]
where $T_{f,\lambda}(x)$ is the Taylor series of $f$ at $\lambda$ (see \hyperref[def:8.2]{Def 8.2}).

Since $f\in C^\infty(I)$, conditions of \hyperref[claim:8.3]{Claim 8.3} are satisfied for $\forall n\in\N$ and, due to \hyperref[eq:8:5:1]{Formula (5)}, we have
\[
f(x)=T_{f,\lambda}^{n}(x)+R_n(x),
\qquad \forall n\in\N\cup\{0\},\ \forall x\in I,
\tag{7}\label{eq:8:7:1}
\]
where
\[
T_{f,\lambda}^{n}(x)=\sum_{k=0}^{n}\frac{f^{(k)}(\lambda)}{k!}(x-\lambda)^k
\]
is the $n$-th partial sum of $T_{f,\lambda}(x)$ and
\[
R_n(x)=\frac{f^{(n+1)}(\beta)}{(n+1)!}(x-\lambda)^{n+1}
\]
for some $\beta$ between $\lambda$ and $x$.

Since, due to \hyperref[eq:8:7:1]{Equality (7)}, for every $x\in I$ we have
\[
\lim_{n\to\infty}R_n(x)
=\lim_{n\to\infty}\bigl(f(x)-T_{f,\lambda}^{n}(x)\bigr)
=f(x)-\lim_{n\to\infty}T_{f,\lambda}^{n}(x),
\]
\hyperref[eq:8:6:1]{Condition (6)} can be written as
\[
\exists\delta>0\quad \forall x\in(\lambda-\delta,\lambda+\delta)
\quad\to\quad \lim_{n\to\infty}R_n(x)=0.
\tag{8}\label{eq:8:8:1}
\]
Let $\delta=d$, where $d>0$ is from \hyperref[item:claim:8.4:2]{Item 2)}, then, $\forall x\in(\lambda-d,\lambda+d)$, we have
\[
|R_n(x)|
\leq C\cdot\frac{(n+1)!}{L^{n+1}}\cdot\frac{1}{(n+1)!}\cdot d^{n+1}
=C\left(\frac{d}{L}\right)^{n+1}.
\tag{9}\label{eq:8:9:1}
\]
Indeed, the first inequality holds due to the definition of $R_n(x)$, since $\beta$ is between $\lambda$ and $x$, hence $\beta\in(\lambda-d,\lambda+d)$, and since $|x-\lambda|<d$.
Since $L>d>0$, we have
\[
\lim_{n\to\infty}C\left(\frac{d}{L}\right)^{n+1}=0.
\]
Therefore, \hyperref[eq:8:9:1]{Inequality (9)} implies that $\lim_{n\to\infty}R_n(x)=0$ for every $x\in(\lambda-d,\lambda+d)$. Thus \hyperref[eq:8:8:1]{Condition (8)} (and \hyperref[eq:8:6:1]{(6)}, since they are equivalent) is proved.

$1)\Rightarrow2)$ First note that if we differentiate $n\in\N\cup\{0\}$ times term-by-term the well-known equality
\[
\sum_{k=0}^{\infty}x^k=\frac{1}{1-x},\qquad \forall x\in(-1,1),
\]
due to \hyperref[claim:7.6]{Claim 7.6}, we will receive
\[
\sum_{k=n}^{\infty}\fall{k}{n}x^{k-n}=\frac{n!}{(1-x)^{n+1}},
\qquad \forall x\in(-1,1),
\tag{10}\label{eq:8:10:1}
\]
where
\[
\fall{k}{n}=
\begin{cases}
k(k-1)\cdots(k-n+1),&n\in\N,\\
1,&n=0.
\end{cases}
\]
Due to \hyperref[item:claim:8.1:2]{Item 2)} of \hyperref[claim:8.1]{Claim 8.1}, $f\in C^\omega(\{\lambda\})$ implies that
\[
f(x)=T_{f,\lambda}(x)
=\sum_{k=0}^{\infty}\frac{f^{(k)}(\lambda)}{k!}(x-\lambda)^k,
\qquad \forall x\in(\lambda-\delta,\lambda+\delta)\subset I
\tag{11}\label{eq:8:11:1}
\]
for some $\delta>0$.
Note that, due to \hyperref[def:7.2]{Def 7.2} and \hyperref[eq:8:11:1]{Equality (11)}, the radius of convergence of the series $T_{f,\lambda}(x)$ is at least $\delta>0$.
Let $p=\frac34\delta>0$, then, due to \hyperref[claim:7.9]{Claim 7.9},
\[
\exists\widehat C>0\quad \forall k\in\N\cup\{0\}
\quad\to\quad
\left|\frac{f^{(k)}(\lambda)}{k!}\right|\leq\frac{\widehat C}{p^k}.
\tag{12}\label{eq:8:12:1}
\]
Let $d=\frac14\delta>0$, then
\[
(\lambda-d,\lambda+d)\subset(\lambda-\delta,\lambda+\delta)\subset I.
\]
If we differentiate $n$ times term-by-term \hyperref[eq:8:11:1]{Equality (11)}, due to \hyperref[claim:7.6]{Claim 7.6}, we will receive
\[
f^{(n)}(x)
=\sum_{k=n}^{\infty}\fall{k}{n}\frac{f^{(k)}(\lambda)}{k!}(x-\lambda)^{k-n},
\qquad \forall x\in(\lambda-\delta,\lambda+\delta),
\tag{13}\label{eq:8:13:1}
\]
where coefficients $\fall{k}{n}$ are defined as in \hyperref[eq:8:10:1]{(10)}.

Let $y\in(\lambda-d,\lambda+d)$.

\begin{center}
\begin{minipage}{\textwidth}
\centering
\includegraphics[width=0.72\textwidth]{figure_interval_diagram.jpg}
\par\smallskip
\refstepcounter{figure}\label{fig:8.1}
{\small Figure \thefigure}
\end{minipage}
\end{center}
Then, for $\forall n\in\N\cup\{0\}$, we have
\begin{align*}
|f^{(n)}(y)|
&\leq \sum_{k=n}^{\infty}\fall{k}{n}\left|\frac{f^{(k)}(\lambda)}{k!}\right||y-\lambda|^{k-n}\\
&\leq \sum_{k=n}^{\infty}\fall{k}{n}\frac{\widehat C}{p^k}|y-\lambda|^{k-n}\\
&=\frac{\widehat C}{p^n}\sum_{k=n}^{\infty}\fall{k}{n}
\left(\frac{|y-\lambda|}{p}\right)^{k-n}\\
&=\frac{\widehat C}{p^n}\cdot
\frac{n!}{\left(1-\frac{|y-\lambda|}{p}\right)^{n+1}}\\
&=\frac{p\widehat C\,n!}{(p-|y-\lambda|)^{n+1}}\\
&\leq \frac{p\widehat C\,n!}{(\frac12\delta)^{n+1}}
=\frac{2p\widehat C}{\delta}\cdot\frac{n!}{(\frac12\delta)^n}
=C\cdot\frac{n!}{L^n},
\end{align*}
where
\[
C=\frac{2p\widehat C}{\delta}
=\frac{2\cdot\frac34\delta\widehat C}{\delta}
=\frac32\widehat C>0,
\qquad
L=\frac12\delta>0.
\]
Here we used \hyperref[eq:8:10:1]{Formula (10)}, $\frac{|y-\lambda|}{p}<\frac{d}{p}=\frac13<1$, and, since $p=\frac34\delta$ and $|y-\lambda|<d=\frac14\delta$, we have $p-|y-\lambda|>\frac12\delta$.
Since $L=\frac12\delta>\frac14\delta=d>0$, the statement is proved.
\end{proof}

\begin{claim}\label{claim:8.5}
\begin{enumerate}
    \item\label{item:claim:8.5:1} the functions $e^x$, $\sin x$, and $\cos x$ are analytic on $\R$;
    \item\label{item:claim:8.5:2} $\ln x$ is analytic on $(0,+\infty)$.
\end{enumerate}
\end{claim}

\begin{proof}
2) Due to \hyperref[claim:8.4]{Claim 8.4}, it suffices to show that for every $\lambda\in(0,+\infty)$ there are constants $L>d>0$ and $C>0$ such that
\[
|((\ln x)^{(k)})|\leq C\cdot\frac{k!}{L^k},
\qquad \forall x\in(\lambda-d,\lambda+d)\subset(0,+\infty).
\]
Let $d=\frac14\lambda$ and $L=\frac12\lambda$. Then
\[
(\lambda-d,\lambda+d)=\left(\frac34\lambda,\frac54\lambda\right)\subset(0,+\infty),
\]
and, for every $k\in\N$ and every $x\in(\frac34\lambda,\frac54\lambda)$, we have
\[
|((\ln x)^{(k)})|
=\frac{(k-1)!}{x^k}
\leq (k-1)!\left(\frac{4}{3\lambda}\right)^k
\leq \frac{k!}{L^k}.
\]
For $k=0$, the function $\ln x$ is bounded on the closed interval
$[\frac34\lambda,\frac54\lambda]$. Hence, choosing
\[
C=\max\left\{1,\ \max_{x\in[\frac34\lambda,\frac54\lambda]}|\ln x|\right\},
\]
we obtain
\[
|((\ln x)^{(k)})|\leq C\cdot\frac{k!}{L^k}
\qquad\forall k\in\N\cup\{0\},\ \forall x\in(\lambda-d,\lambda+d),
\]
where $L=\frac12\lambda>d=\frac14\lambda>0$.

1) We leave it as exercise (similar to \hyperref[item:claim:8.5:2]{Item 2)}).
\end{proof}

We will use the following claim to construct an important example of a smooth non-analytic function.

\begin{claim}[The Borel Lemma]\label{claim:8.6}
For every sequence $\{\alpha_n\}_{n=0}^{\infty}$ there exists a smooth function $f$ on $\R$ such that
\[
f^{(n)}(0)=\alpha_n,
\qquad \forall n\in\N\cup\{0\}.
\]
\end{claim}

\begin{proof}
For $\forall x\in\R$ let us consider the following functions
\[
\widetilde\tau(x)=
\begin{cases}
e^{-1/x},&x>0,\\0,&x\leq0,
\end{cases}
\qquad
 g(x):=\frac{\widetilde\tau(x)}{\widetilde\tau(x)+\widetilde\tau(1-x)},
\qquad
 h(x):=g(2+x)\,g(2-x).
\]

$\widetilde\tau\in C^\infty(\R)$ due to Item 2 of \hyperref[claim:8.2]{Claim 8.2}.

$g\in C^\infty(\R)$ since $\widetilde\tau(x)+\widetilde\tau(1-x)>0$, $\forall x\in\R$, and $\widetilde\tau\in C^\infty(\R)$.

$h\in C^\infty(\R)$ since $g\in C^\infty(\R)$.

It ought to be clear that the function $h(x)$ equals $1$ on the closed interval $[-1,1]$ and vanishes outside the open interval $(-2,2)$. In particular,
\[
h^{(n)}(0)=0,\qquad \forall n\in\N.
\]
For $\forall k\in\N\cup\{0\}$, $\forall x\in\R$, let
\[
u_k(x):=x^k h(x).
\]
Then it is clear that, $\forall k\in\N\cup\{0\}$, we have
\begin{itemize}
    \item $u_k(x)=x^k$, $\forall x\in[-1,1]$, and $u_k(x)=0$, $\forall x\in\R\setminus(-2,2)$;
    \item $u_k(x)\in C^\infty(\R)$;
    \item
    \[
    u_k^{(n)}(0)=
    \begin{cases}
n!,&n=k,\\0,&n\neq k,
    \end{cases}
    \qquad \forall n\in\N\cup\{0\},
    \tag{14}\label{eq:8:14:1}
    \]
    where $u_k^{(n)}(0)$ is the $n$-th derivative of $u_k$ at $x=0$.
\end{itemize}

$\forall n\in\N\cup\{0\}$ the function $u_k^{(n)}(x)$ is bounded on $\R$, that is
\[
\forall k,n\in\N\cup\{0\}\quad \exists L_{k,n}>0\quad \forall x\in\R
\quad\to\quad |u_k^{(n)}(x)|\leq L_{k,n}.
\tag{15}\label{eq:8:15:1}
\]
Indeed, $\forall n\in\N\cup\{0\}$ we have $u_k^{(n)}(x)=0$, $\forall x\in\R\setminus(-2,2)$, thus it suffices to prove that $u_k^{(n)}(x)$ is bounded on $[-2,2]$. Since $u_k^{(n)}$ is continuous on $[-2,2]$, the claim follows from the fact that a continuous on a closed interval function attains a maximum and a minimum on the interval (see Calculus-1).

Let
\[
M_k=\max\{1,|\alpha_k|,L_{k,0},L_{k,1},\ldots,L_{k,k}\},
\qquad \forall k\in\N\cup\{0\},
\tag{16}\label{eq:8:16:1}
\]
where $\alpha_k$ is the $k$-th element of the sequence $\{\alpha_k\}_{k=0}^{\infty}$ and constants $L_{k,i}$ are defined in \hyperref[eq:8:15:1]{(15)}.

For $\forall k\in\N\cup\{0\}$, $\forall x\in\R$, let
\[
f_k(x)=\frac{\alpha_k}{(M_k)^k\,k!}\,u_k(M_kx).
\]
Then we have
\[
f_k^{(n)}(x)
=\frac{\alpha_k}{(M_k)^{k-n}\,k!}\,u_k^{(n)}(M_kx),
\qquad \forall k,n\in\N\cup\{0\},\ \forall x\in\R,
\tag{17}\label{eq:8:17:1}
\]
and, due to \hyperref[eq:8:14:1]{Equality (14)},
\[
f_k^{(n)}(0)=
\begin{cases}
\alpha_n,&n=k,\\0,&n\neq k,
\end{cases}
\qquad \forall k,n\in\N\cup\{0\}.
\]

Note that, $\forall n\in\N\cup\{0\}$, $\forall x\in\R$, we have
\begin{align*}
\left|\sum_{k=0}^{\infty}f_k^{(n)}(x)\right|
&\leq\sum_{k=0}^{\infty}|f_k^{(n)}(x)|
\leq\sum_{k=0}^{\infty}\frac{|\alpha_k|}{M_k^{k-n}\,k!}L_{k,n}\\
&=\sum_{k=0}^{n+1}\frac{|\alpha_k|}{M_k^{k-n}\,k!}L_{k,n}
+\sum_{k=n+2}^{\infty}\frac1{k!}\cdot\frac{|\alpha_k|}{M_k}\cdot\frac{L_{k,n}}{M_k}\cdot\frac1{M_k^{k-n-2}}\\
&\leq C_n+\sum_{k=n+2}^{\infty}\frac1{k!},
\tag{18}\label{eq:8:18:1}
\end{align*}
where $C_n\in\R$.
Here we used $|\alpha_k|/M_k\in[0,1]$, $L_{k,n}/M_k\in[0,1]$ for $k>n$, and $1/M_k^{k-n-2}\in(0,1]$ for $k\geq n+2$, due to \hyperref[eq:8:16:1]{(16)}.

Since the series $\sum_{k=n+2}^{\infty}\frac1{k!}$ converges (due to \hyperref[claim:2.11]{Claim 2.11}), due to \hyperref[eq:8:18:1]{Inequality (18)} and \hyperref[claim:5.7]{Claim 5.7} [The Weierstrass M-test], $\forall n\in\N\cup\{0\}$ the functional series $\sum_{k=0}^{\infty}f_k^{(n)}(x)$ converges uniformly on $\R$.
Therefore, due to \hyperref[claim:6.13]{Claim 6.13}, we have
\[
f^{(n)}(x)=\sum_{k=0}^{\infty}f_k^{(n)}(x),
\qquad \forall n\in\N,\ \forall x\in\R,
\tag{19}\label{eq:8:19:1}
\]
where we defined
\[
f(x):=\sum_{k=0}^{\infty}f_k(x),
\qquad \forall x\in\R.
\]
It remains to note that:
\begin{itemize}
    \item $f\in C^\infty(\R)$ due to \hyperref[eq:8:19:1]{(19)};
    \item
    \[
    f^{(n)}(0)=\sum_{k=0}^{\infty}f_k^{(n)}(0)=\alpha_n,
    \qquad \forall n\in\N\cup\{0\}.
    \]
\end{itemize}
\end{proof}

\begin{claim}[Pathological Examples of Smooth Functions]\label{claim:8.7}
There exists a smooth function $f$ on $\R$ such that
\begin{enumerate}
    \item\label{item:claim:8.7:1} its Taylor series $T_{f,0}(x)$ has zero radius of convergence;
    \item\label{item:claim:8.7:2} it is not analytic at any point.
\end{enumerate}
\end{claim}

\begin{proof}
1) Let $a_0=1$, $a_k=k!\,k^k$, $\forall k\in\N$.
Then, due to \hyperref[claim:8.6]{Claim 8.6}, there is $f\in C^\infty(\R)$ such that
\[
f^{(k)}(0)=a_k,\qquad \forall k\in\N\cup\{0\}.
\]
Then, due to \hyperref[def:8.2]{Def 8.2}, we have
\[
T_{f,0}(x)
=\sum_{k=0}^{\infty}\frac{f^{(k)}(0)}{k!}(x-0)^k
=\sum_{k=0}^{\infty}k^k x^k.
\]
It is clear that the radius of convergence of the series $\sum_{k=0}^{\infty}k^kx^k$ is equal to zero (see \hyperref[claim:7.2]{Claim 7.2}).

2) Let
\[
f(x)=\sum_{k=1}^{\infty}\cos(2^k x)\,e^{-\sqrt{2^k}},
\qquad \forall x\in\R.
\]
Then it can be shown that $f\in C^\infty(\R)$ (this is not difficult and we leave it as an exercise) and $f$ is nowhere analytic on $\R$ (the proof is difficult, so we shall merely state the fact).
\end{proof}
